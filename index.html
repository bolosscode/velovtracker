#!/usr/bin/env python3 """ bootstrap.py — Import historique Vélo'v par tranches d'1 jour (CSV). Format événementiel pur (timestamp exact) + tableau positionnel + gzip. Format delta : base: {"t": "...", "s": [[n,b,s,e,m], ...]} deltas: [{"t": "...", "d": [[n,b_or_null,...], ...]}, ...] """ import os, sys, json, gzip, argparse, time, csv, io, ast from datetime import datetime, timezone, date, timedelta from collections import defaultdict from concurrent.futures import ThreadPoolExecutor, as_completed import requests PAGE_SIZE = 500000 CHUNK_DAYS = 1 WORKERS = 10 STATE_FILE = 'data/bootstrap_state.json' FLUSH_INTERVAL = 300 BASE_URL = ( "https://data.grandlyon.com/fr/datapusher/ws/timeseries" "/jcd_jcdecaux.historiquevelov/all.csv" "?maxfeatures={size}&start={start}" "&horodate__gte={gte}&horodate__lt={lt}" ) parser = argparse.ArgumentParser() parser.add_argument("--user", default="") parser.add_argument("--password", default="") parser.add_argument("--out", default="data/history") parser.add_argument("--from-date", default="2025-01-29", dest="from_date") parser.add_argument("--to-date", default=str(date.today()), dest="to_date") args = parser.parse_args() auth = (args.user, args.password) if args.user else None os.makedirs(args.out, exist_ok=True) print(f"Import {args.from_date} → {args.to_date} | {WORKERS} workers | événementiel+gzip", flush=True) VAE_LAUNCH = '2025-01-29' # ── Helpers ──────────────────────────────────────────────────────────────────── def parse_stands(raw): if not raw or raw.strip() in ('', '""', "''"): return {} try: return ast.literal_eval(raw.strip().strip('"')) except: pass try: return json.loads(raw.strip().strip('"').replace("'", '"')) except: return {} def safe_int(v): try: return int(v or 0) except: return 0 def parse_row(row, today): """Parse une ligne CSV → [n, b, s, e, m]""" tot = parse_stands(row.get('total_stands','') or row.get('main_stands','')) tav = tot.get('availabilities', {}) if isinstance(tot, dict) else {} bikes = safe_int(tav.get('bikes')) cap = safe_int(tot.get('capacity') if isinstance(tot, dict) else 0) stands = (cap - bikes) if cap else safe_int(tav.get('stands')) if today >= VAE_LAUNCH: elec = safe_int(tav.get('electricalInternalBatteryBikes')) meca = safe_int(tav.get('mechanicalBikes')) + safe_int(tav.get('electricalRemovableBatteryBikes')) else: elec = 0 meca = bikes num = row.get('number') or row.get('num') or row.get('id') if not num: return None, None ts = row.get('horodate','').strip() if not ts: return None, None return ts, [safe_int(num), bikes, stands, elec, meca] # ── Compression delta en format tableau ──────────────────────────────────────── FIELDS_IDX = [1, 2, 3, 4] # indices b, s, e, m dans le tableau def build_delta_file(events): """ events : dict {ts_str: {num: [n,b,s,e,m]}} Retourne un dict {base, deltas} en format tableau positionnel. """ if not events: return None timestamps = sorted(events.keys()) base_ts = timestamps[0] base_stations = list(events[base_ts].values()) prev = {st[0]: st[:] for st in base_stations} deltas = [] for ts in timestamps[1:]: curr = events[ts] changed = [] for n, st in curr.items(): p = prev.get(n) if p is None: changed.append(st) else: diff = [n, None, None, None, None] has_change = False for i in FIELDS_IDX: if st[i] != p[i]: diff[i] = st[i] has_change = True if has_change: changed.append(diff) prev[n] = st # Stations disparues for n in list(prev.keys()): if n not in curr: changed.append([n, None, None, None, None, 'CLOSED']) del prev[n] if changed: deltas.append({'t': ts, 'd': changed}) return {'base': {'t': base_ts, 's': base_stations}, 'deltas': deltas} # ── Fetch ────────────────────────────────────────────────────────────────────── def fetch_day(day_date): gte = day_date.isoformat() lt = (day_date + timedelta(days=1)).isoformat() today = day_date.isoformat() for attempt in range(3): try: page_start = 1 events = defaultdict(dict) # {ts: {num: [n,b,s,e,m]}} total_rows = 0 while True: url = BASE_URL.format(size=PAGE_SIZE, start=page_start, gte=gte, lt=lt) r = requests.get(url, auth=auth, timeout=60) r.raise_for_status() reader = csv.DictReader(io.StringIO(r.text.lstrip('\ufeff')), delimiter=';') rows = list(reader) for row in rows: ts, station = parse_row(row, today) if ts and station: events[ts][station[0]] = station total_rows += len(rows) if len(rows) < PAGE_SIZE: break page_start += PAGE_SIZE return day_date, events, total_rows except Exception as e: if attempt == 2: print(f" ÉCHEC {day_date}: {e}", flush=True) return day_date, {}, 0 time.sleep(180) # ── Flush ────────────────────────────────────────────────────────────────────── def flush_day(day_date, events): if not events: return False path = os.path.join(args.out, f'{day_date}.json.gz') data = build_delta_file(events) if not data: return False with gzip.open(path, 'wt', encoding='utf-8', compresslevel=9) as f: json.dump(data, f, ensure_ascii=False, separators=(',', ':')) size_kb = os.path.getsize(path) // 1024 print(f" ✓ {day_date} → {size_kb} KB gz | {len(events)} ts | {len(data['deltas'])} deltas", flush=True) return True # ── State ────────────────────────────────────────────────────────────────────── def save_state(current_date, finished=False): os.makedirs(os.path.dirname(STATE_FILE), exist_ok=True) with open(STATE_FILE, 'w') as f: json.dump({'current_date': str(current_date), 'finished': finished, 'updated_at': datetime.now(timezone.utc).isoformat(timespec='seconds')}, f) def load_state(): if os.path.exists(STATE_FILE): try: s = json.load(open(STATE_FILE)) if s.get('finished'): print("✅ Déjà complet."); sys.exit(0) d = max(s.get('current_date', args.from_date), args.from_date) print(f"Reprise depuis : {d}"); return d except: pass return args.from_date # ── Main ─────────────────────────────────────────────────────────────────────── current = date.fromisoformat(load_state()) end = date.fromisoformat(args.to_date) total = 0 flushed = 0 t0 = last_save = time.time() days_to_fetch = [] d = current while d < end: days_to_fetch.append(d) d += timedelta(days=1) print(f"Démarrage à {current} | {len(days_to_fetch)} jours à fetcher", flush=True) for i in range(0, len(days_to_fetch), WORKERS): batch = days_to_fetch[i:i+WORKERS] with ThreadPoolExecutor(max_workers=WORKERS) as ex: futures = {ex.submit(fetch_day, day): day for day in batch} results = {} for fut in as_completed(futures): day_date, events, n_rows = fut.result() results[day_date] = (events, n_rows) total += n_rows elapsed = time.time() - t0 print(f" {day_date} | {n_rows:,} lignes | {len(events)} ts | total={total:,} | {total/max(1,elapsed):,.0f} l/s", flush=True) # Flush dans l'ordre for day_date in sorted(results.keys()): events, _ = results[day_date] if flush_day(day_date, events): flushed += 1 current = batch[-1] + timedelta(days=1) if time.time() - last_save >= FLUSH_INTERVAL: save_state(current) last_save = time.time() finished = current >= end save_state(current, finished=finished) elapsed = time.time() - t0 print(f"\n✓ {total:,} lignes en {elapsed:.0f}s ({total/max(1,elapsed):.0f} l/s) → {flushed} jours", flush=True) print("✅ TERMINÉ." if finished else f"⏸ PARTIEL — reprise à {current}.", flush=True)
